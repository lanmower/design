export function FileRow({ name, type, size, modified, onOpen }) {
    return h('div', {
        class: 'ds-file-row',
        'data-file-type': type,
        onclick: onOpen
    },
        FileIcon({ type }),
        h('span', { class: 'title' }, name),
        h('span', { class: 'meta' }, fmtFileSize(size))
    );
}
